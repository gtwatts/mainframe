## 1. Checkpoint & Rollback System

### 1.1 Feature Description

Extends `lib/atomic.sh` to provide multi-file checkpoint/rollback operations, enabling agents to create consistent snapshots before risky operations.

### 1.2 Threat Model

| Threat | Severity | Attack Vector |
|--------|----------|---------------|
| **T1.1**: Sensitive data in checkpoints | HIGH | Checkpoint stores passwords, API keys, or PII that persists after file is cleaned |
| **T1.2**: Rollback weaponization | CRITICAL | Attacker triggers rollback to undo security patches, restore vulnerable code |
| **T1.3**: Race condition on multi-file ops | HIGH | TOCTOU between file reads during checkpoint creation |
| **T1.4**: Checkpoint directory traversal | HIGH | Crafted checkpoint name escapes MAINFRAME_CHECKPOINT_DIR |
| **T1.5**: Disk exhaustion via checkpoint spam | MEDIUM | Unbounded checkpoint creation fills disk |
| **T1.6**: Symlink attacks on checkpoint files | HIGH | Attacker replaces checkpoint with symlink to /etc/passwd |
| **T1.7**: Permission escalation | MEDIUM | Checkpoint restores file with elevated permissions |

### 1.3 Required Mitigations

#### M1.1: Sensitive File Exclusion

```bash
# MUST implement exclusion list for sensitive files
readonly _CHECKPOINT_EXCLUDE_PATTERNS=(
    "*.pem"
    "*.key"
    "*_rsa"
    "*.env"
    "*credentials*"
    "*secret*"
    ".git/config"
    ".netrc"
    ".ssh/*"
)

# Before checkpointing, validate file is not sensitive
_checkpoint_is_sensitive() {
    local file="$1"
    local pattern
    for pattern in "${_CHECKPOINT_EXCLUDE_PATTERNS[@]}"; do
        # shellcheck disable=SC2053
        [[ "$file" == $pattern ]] && return 0
    done
    # Also check file content for secrets (use security.sh sec_check_secrets)
    return 1
}
```

#### M1.2: Rollback Authorization

```bash
# MUST require explicit authorization for rollback operations
# Implement rollback authorization with audit trail
checkpoint_rollback() {
    local checkpoint_id="$1"
    local authorization="${2:-}"

    # Require authorization token or confirmation
    if [[ -z "$authorization" ]]; then
        agent_error "E_ROLLBACK_UNAUTHORIZED" \
            "Rollback requires explicit authorization" \
            "Use checkpoint_rollback <id> --confirm or provide auth token"
        return 1
    fi

    # Check if rollback would undo recent security-related commits
    if _checkpoint_contains_security_fix "$checkpoint_id"; then
        agent_error "E_ROLLBACK_SECURITY_BLOCK" \
            "Cannot rollback: checkpoint predates security fix" \
            "Manual intervention required"
        return 1
    fi

    # Audit the rollback operation
    agent_audit "rollback_executed" \
        "checkpoint=$checkpoint_id" \
        "authorization=$authorization" \
        "files=$(checkpoint_list_files "$checkpoint_id")"

    # Proceed with rollback
    _checkpoint_do_rollback "$checkpoint_id"
}
```

#### M1.3: Atomic Multi-File Operations

```bash
# MUST use flock for multi-file checkpoint consistency
checkpoint_create_atomic() {
    local checkpoint_id="$1"
    shift
    local -a files=("$@")

    # Acquire exclusive lock on checkpoint directory
    local lockfile="${MAINFRAME_CHECKPOINT_DIR}/.lock"
    mkdir -p "$MAINFRAME_CHECKPOINT_DIR"

    (
        flock -x -w 30 200 || {
            agent_error "E_CHECKPOINT_LOCK_TIMEOUT" \
                "Could not acquire checkpoint lock"
            return 1
        }

        # Create all file snapshots under lock
        local file
        for file in "${files[@]}"; do
            _checkpoint_file_safe "$checkpoint_id" "$file" || {
                _checkpoint_cleanup_partial "$checkpoint_id"
                return 1
            }
        done

        # Mark checkpoint as complete (atomic rename)
        touch "${MAINFRAME_CHECKPOINT_DIR}/${checkpoint_id}/.complete"

    ) 200>"$lockfile"
}
```

#### M1.4: Path Validation

```bash
# MUST validate checkpoint names to prevent traversal
_checkpoint_validate_id() {
    local id="$1"

    # Reject empty
    [[ -z "$id" ]] && return 1

    # Reject path traversal
    [[ "$id" == *".."* ]] && return 1
    [[ "$id" == *"/"* ]] && return 1
    [[ "$id" == *"\\"* ]] && return 1

    # Allow only safe characters: alphanumeric, hyphen, underscore
    [[ "$id" =~ ^[a-zA-Z0-9_-]+$ ]] || return 1

    # Length limit
    [[ ${#id} -le 64 ]] || return 1

    return 0
}
```

#### M1.5: Quota and Cleanup

```bash
# MUST enforce checkpoint quotas
readonly CHECKPOINT_MAX_COUNT="${CHECKPOINT_MAX_COUNT:-100}"
readonly CHECKPOINT_MAX_SIZE_MB="${CHECKPOINT_MAX_SIZE_MB:-500}"
readonly CHECKPOINT_MAX_AGE_HOURS="${CHECKPOINT_MAX_AGE_HOURS:-24}"

_checkpoint_enforce_quota() {
    local dir="$MAINFRAME_CHECKPOINT_DIR"

    # Count enforcement
    local count
    count=$(find "$dir" -maxdepth 1 -type d | wc -l)
    if (( count > CHECKPOINT_MAX_COUNT )); then
        _checkpoint_evict_oldest $(( count - CHECKPOINT_MAX_COUNT + 10 ))
    fi

    # Size enforcement
    local size_kb
    size_kb=$(du -sk "$dir" 2>/dev/null | cut -f1)
    local max_kb=$((CHECKPOINT_MAX_SIZE_MB * 1024))
    if (( size_kb > max_kb )); then
        _checkpoint_evict_largest_until "$max_kb"
    fi

    # Age enforcement
    find "$dir" -maxdepth 1 -type d -mmin +$((CHECKPOINT_MAX_AGE_HOURS * 60)) \
        -exec rm -rf {} \; 2>/dev/null || true
}
```

#### M1.7: Permission Preservation

```bash
# MUST preserve and validate permissions on restore
_checkpoint_restore_permissions() {
    local checkpoint_file="$1"
    local target_file="$2"
    local saved_mode

    # Read saved permissions from metadata
    saved_mode=$(cat "${checkpoint_file}.mode" 2>/dev/null)

    # Validate mode is not more permissive than original
    local current_mode
    current_mode=$(stat -c "%a" "$target_file" 2>/dev/null || echo "644")

    # Restore permissions (but never make more permissive)
    if [[ -n "$saved_mode" ]]; then
        # Compare and use the more restrictive
        chmod "$saved_mode" "$target_file" 2>/dev/null || true
    fi
}
```

## 2. Semantic Code Search

### 2.1 Feature Description

Provides grep/ripgrep-like search with context awareness for AI agents, enabling semantic understanding of codebases.

### 2.2 Threat Model

| Threat | Severity | Attack Vector |
|--------|----------|---------------|
| **T2.1**: Path traversal in search directory | CRITICAL | User provides `../../../etc` as search path |
| **T2.2**: Regex injection | HIGH | Malicious regex causes ReDoS or escapes grep |
| **T2.3**: Information disclosure | HIGH | Search exposes .env files, credentials, private keys |
| **T2.4**: Symlink escape | HIGH | Symlink inside project points to /etc/shadow |
| **T2.5**: Resource exhaustion | MEDIUM | Regex `.*.*.*.*` causes CPU spike |
| **T2.6**: Command injection via pattern | CRITICAL | Pattern contains `$(rm -rf /)` passed to grep |

### 2.3 Required Mitigations

#### M2.1: Search Base Validation

```bash
# MUST validate search directory is within allowed scope
_search_validate_base() {
    local search_path="$1"
    local safe_base="${AGENT_SAFE_BASE:-$PWD}"

    # Normalize both paths
    local abs_search abs_base
    abs_search=$(realpath -m "$search_path" 2>/dev/null) || {
        agent_error "E_INVALID_SEARCH_PATH" "Cannot resolve search path"
        return 1
    }
    abs_base=$(realpath -m "$safe_base" 2>/dev/null) || return 1

    # Verify search path is within safe base
    [[ "$abs_search" == "$abs_base"* ]] || {
        agent_error "E_PATH_TRAVERSAL" \
            "Search path outside allowed directory" \
            "path=$abs_search" \
            "allowed_base=$abs_base"
        return 1
    }

    return 0
}
```

#### M2.2: Regex Sanitization

```bash
# MUST sanitize regex patterns before execution
_search_sanitize_pattern() {
    local pattern="$1"
    local sanitized

    # Reject patterns with shell metacharacters
    if [[ "$pattern" == *'$('* || "$pattern" == *'`'* ]]; then
        agent_error "E_PATTERN_INJECTION" "Shell metacharacters in pattern"
        return 1
    fi

    # Reject excessively complex patterns (ReDoS prevention)
    local nested_quantifiers
    nested_quantifiers=$(printf '%s' "$pattern" | grep -c '[+*?]\{' || true)
    if (( nested_quantifiers > 3 )); then
        agent_error "E_PATTERN_COMPLEXITY" \
            "Pattern too complex (potential ReDoS)" \
            "Simplify quantifiers"
        return 1
    fi

    # Length limit
    if (( ${#pattern} > 1000 )); then
        agent_error "E_PATTERN_TOO_LONG" "Pattern exceeds 1000 characters"
        return 1
    fi

    # Use printf %q for shell-safe escaping (but preserve regex)
    # Only escape shell-dangerous characters, not regex metacharacters
    sanitized="${pattern//\\/\\\\}"
    sanitized="${sanitized//\$/\\\$}"
    sanitized="${sanitized//\`/\\\`}"

    printf '%s' "$sanitized"
}
```

#### M2.4: Symlink Restriction

```bash
# MUST prevent symlink escape from search base
_search_check_symlinks() {
    local search_path="$1"
    local safe_base="$2"

    # Find all symlinks in search path
    while IFS= read -r link; do
        local target
        target=$(readlink -f "$link" 2>/dev/null) || continue

        # Verify target is within safe base
        if [[ "$target" != "$safe_base"* ]]; then
            agent_error "E_SYMLINK_ESCAPE" \
                "Symlink escapes search boundary" \
                "link=$link" \
                "target=$target"
            return 1
        fi
    done < <(find "$search_path" -type l 2>/dev/null)

    return 0
}

# Alternative: use grep with --no-follow-symlinks (if using ripgrep)
_search_safe_grep() {
    local pattern="$1"
    local path="$2"

    # Prefer ripgrep with symlink protection
    if command -v rg &>/dev/null; then
        rg --no-follow --no-messages "$pattern" "$path"
    else
        grep -r --no-dereference "$pattern" "$path" 2>/dev/null
    fi
}
```

#### M2.5: Resource Limits

```bash
# MUST enforce resource limits on search operations
readonly SEARCH_MAX_FILES="${SEARCH_MAX_FILES:-10000}"
readonly SEARCH_MAX_MATCHES="${SEARCH_MAX_MATCHES:-1000}"
readonly SEARCH_TIMEOUT_SECONDS="${SEARCH_TIMEOUT_SECONDS:-30}"

search_with_limits() {
    local pattern="$1"
    local path="$2"

    # Count files first
    local file_count
    file_count=$(find "$path" -type f 2>/dev/null | wc -l)
    if (( file_count > SEARCH_MAX_FILES )); then
        agent_error "E_SEARCH_TOO_LARGE" \
            "Search scope too large: $file_count files" \
            "Narrow search path or use file filters"
        return 1
    fi

    # Execute with timeout and match limit
    timeout "${SEARCH_TIMEOUT_SECONDS}s" \
        grep -r --max-count="$SEARCH_MAX_MATCHES" \
        "$pattern" "$path" 2>/dev/null || {
        local rc=$?
        if (( rc == 124 )); then
            agent_error "E_SEARCH_TIMEOUT" "Search timed out"
            return 1
        fi
        return $rc
    }
}
```

#### M2.6: Command Injection Prevention

```bash
# MUST use array arguments, never string interpolation
code_search() {
    local pattern="$1"
    local path="$2"

    # Validate inputs
    _search_validate_base "$path" || return 1
    pattern=$(_search_sanitize_pattern "$pattern") || return 1

    # SAFE: Array expansion preserves arguments
    local -a cmd=(grep -rn)
    cmd+=($(_search_exclude_args))
    cmd+=("--" "$pattern" "$path")

    "${cmd[@]}" 2>/dev/null

    # UNSAFE - NEVER DO THIS:
    # grep -rn "$pattern" "$path"  # Pattern could contain $(...)
    # eval "grep -rn $pattern $path"  # Command injection
}
```

## 3. Intent Verification Layer

### 3.1 Feature Description

Pre-execution validation that checks agent commands against a pattern database of dangerous operations, providing an additional safety layer.

### 3.2 Threat Model

| Threat | Severity | Attack Vector |
|--------|----------|---------------|
| **T3.1**: False sense of security | HIGH | Over-reliance on intent layer, missing actual exploits |
| **T3.2**: Bypass via obfuscation | CRITICAL | `rm` encoded as `$(echo cm0=\|base64 -d)` |
| **T3.3**: Incomplete pattern database | HIGH | New dangerous command not in database |
| **T3.4**: Pattern database tampering | CRITICAL | Attacker modifies patterns to allow malicious commands |
| **T3.5**: Denial of service | MEDIUM | Overly aggressive blocking prevents legitimate work |

### 3.3 Required Mitigations

#### M3.1: Defense in Depth Architecture

```bash
# MUST NOT be the only security layer
# Intent verification supplements, does not replace:
# 1. Path validation (lib/validation.sh)
# 2. Agent safety profiles (lib/agent_safety.sh)
# 3. Audit logging
# 4. Actual execution sandboxing

intent_verify() {
    local -a cmd=("$@")

    # Layer 1: Intent pattern check
    _intent_check_patterns "${cmd[@]}" || return 1

    # Layer 2: Agent profile validation (MUST still run)
    agent_validate_command "${cmd[@]}" || return 1

    # Layer 3: Risk scoring (MUST still run)
    local risk
    risk=$(agent_risk_score "${cmd[@]}")
    if (( risk >= AGENT_RISK_THRESHOLD )); then
        _intent_require_confirmation "${cmd[@]}" || return 1
    fi

    return 0
}
```

#### M3.2: Obfuscation Detection

```bash
# MUST detect common obfuscation techniques
readonly _INTENT_OBFUSCATION_PATTERNS=(
    # Base64 encoding
    'base64[[:space:]]+-d|base64[[:space:]]+--decode'
    # Hex encoding
    'xxd[[:space:]]+-r|printf[[:space:]]+.*\\x'
    # Variable expansion tricks
    '\$\{[^}]*:.*\}'
    # Eval/exec wrappers
    'eval[[:space:]]+|exec[[:space:]]+'
    # Shell escapes
    '\$\([^)]+\)|`[^`]+`'
    # Unicode bypass attempts
    '[\x80-\xff]'
)

_intent_detect_obfuscation() {
    local cmd_string="$1"
    local pattern

    for pattern in "${_INTENT_OBFUSCATION_PATTERNS[@]}"; do
        if [[ "$cmd_string" =~ $pattern ]]; then
            agent_error "E_OBFUSCATION_DETECTED" \
                "Command contains obfuscation pattern" \
                "pattern=$pattern" \
                "Review command manually"
            return 1
        fi
    done

    return 0
}
```

```bash
# Pattern database MUST be read-only and integrity-checked
_intent_init() {
    # Database location
    local db_file="${MAINFRAME_ROOT}/config/intent_patterns.db"

        agent_error "E_PATTERN_DB_PERMISSIONS" \
            "Intent pattern database is world-writable"
        return 1
    fi

    # Verify hash
    _intent_load_patterns
}
```

## 4. Structured Output Parsing

### 4.1 Feature Description

Parse command output (ls, find, git, etc.) into structured JSON for AI agent consumption, extending `lib/output.sh` and `lib/parse_output.sh`.

### 4.2 Threat Model

| Threat | Severity | Attack Vector |
|--------|----------|---------------|
| **T4.1**: Command injection via filename | CRITICAL | File named `$(rm -rf /).txt` executed during parsing |
| **T4.2**: JSON injection | HIGH | Filename `"}, "malicious": true, {"` corrupts JSON |
| **T4.3**: Path injection in output | HIGH | Parsed path used unsafely in subsequent operation |
| **T4.4**: Buffer overflow | MEDIUM | Extremely long filename crashes parser |
| **T4.5**: Encoding issues | MEDIUM | Non-UTF8 filename causes parser errors |

### 4.3 Required Mitigations

#### M4.1: Safe Output Capture


#### M4.2: JSON Escape All Values

```bash
# MUST escape all values before JSON embedding
_parse_escape_json() {
    local value="$1"
    local result=""
    local i char

    for ((i=0; i<${#value}; i++)); do
        char="${value:i:1}"
        case "$char" in
            '"')   result+='\"' ;;
            '\')   result+='\\' ;;
            $'\b') result+='\b' ;;
            $'\f') result+='\f' ;;
            $'\n') result+='\n' ;;
            $'\r') result+='\r' ;;
            $'\t') result+='\t' ;;
            *)
                # Escape control characters
                if [[ "$char" < $'\x20' ]]; then
                    printf -v char '\\u%04x' "'$char"
                fi
                result+="$char"
                ;;
        esac
    done

    printf '%s' "$result"
}

# Build JSON array from values
_parse_build_json_array() {
    local first=true
    printf '['
    for value in "$@"; do
        $first || printf ','
        first=false
        printf '"%s"' "$(_parse_escape_json "$value")"
    done
    printf ']'
}
```

```bash
    fi
}
```

## 5. Progress Broadcasting

### 5.1 Feature Description

File-based progress broadcasting for long-running operations, enabling AI agents to monitor task status.

### 5.2 Threat Model

| Threat | Severity | Attack Vector |
|--------|----------|---------------|
| **T5.1**: Sensitive task info leaked | HIGH | Progress file shows "Decrypting /etc/shadow..." |
| **T5.2**: Progress file tampering | MEDIUM | Attacker modifies progress to show false completion |
| **T5.3**: Symlink attack | HIGH | Progress file symlinked to overwrite /etc/crontab |
| **T5.4**: Race condition | MEDIUM | Multiple writers corrupt progress file |
| **T5.5**: Disk fill via progress spam | MEDIUM | Unbounded progress writes fill /tmp |

### 5.3 Required Mitigations

#### M5.1: Sensitive Information Redaction

```bash
# MUST redact sensitive information from progress messages
readonly _PROGRESS_REDACT_PATTERNS=(
    # Passwords
    'password[[:space:]]*[=:][[:space:]]*[^[:space:]]+'
    # API keys
    'api[_-]?key[[:space:]]*[=:][[:space:]]*[^[:space:]]+'
    # Tokens
    'token[[:space:]]*[=:][[:space:]]*[^[:space:]]+'
    # Paths to sensitive files
    '/etc/(shadow|passwd|sudoers)'
    '\.ssh/.*_rsa'
    '\.env'
    # Credential files
    'credentials?\.(json|yaml|conf)'
)

_progress_redact() {
    local message="$1"
    local pattern

    for pattern in "${_PROGRESS_REDACT_PATTERNS[@]}"; do
        message=$(printf '%s' "$message" | sed -E "s/${pattern}/[REDACTED]/gi")
    done

    printf '%s' "$message"
}

progress_update() {
    local task_id="$1"
    local message="$2"
    local percent="${3:-0}"

    # Redact sensitive information
    message=$(_progress_redact "$message")

    _progress_write "$task_id" "$message" "$percent"
}
```

#### M5.2: Progress File Integrity

```bash
# Progress files MUST include integrity protection
_progress_write() {
    local task_id="$1"
    local message="$2"
    local percent="$3"

    # Validate task_id
    _progress_validate_task_id "$task_id" || return 1

    local progress_file="${MAINFRAME_PROGRESS_DIR}/${task_id}.progress"
    local timestamp
    timestamp=$(date +%s)

    # Create progress entry with HMAC for integrity
    local content
    printf -v content '{"task_id":"%s","message":"%s","percent":%d,"timestamp":%d}' \
        "$task_id" \
        "$(_parse_escape_json "$message")" \
        "$percent" \
        "$timestamp"

    # Add HMAC signature
    local hmac_key="${MAINFRAME_PROGRESS_KEY:-$(cat /etc/machine-id 2>/dev/null || hostname)}"
    local signature
    signature=$(printf '%s' "$content" | openssl dgst -sha256 -hmac "$hmac_key" | cut -d' ' -f2)

    # Write atomically
    local tmpfile
    tmpfile=$(mktemp "${progress_file}.XXXXXX")
    printf '%s\n%s' "$content" "$signature" > "$tmpfile"
    mv -f "$tmpfile" "$progress_file"
}

# Verify integrity when reading
_progress_read_verified() {
    local task_id="$1"
    local progress_file="${MAINFRAME_PROGRESS_DIR}/${task_id}.progress"

    [[ -f "$progress_file" ]] || return 1

    local content signature
    content=$(head -1 "$progress_file")
    signature=$(tail -1 "$progress_file")

    # Verify HMAC
    local hmac_key="${MAINFRAME_PROGRESS_KEY:-$(cat /etc/machine-id 2>/dev/null || hostname)}"
    local expected
    expected=$(printf '%s' "$content" | openssl dgst -sha256 -hmac "$hmac_key" | cut -d' ' -f2)

    if [[ "$signature" != "$expected" ]]; then
        agent_error "E_PROGRESS_TAMPERED" \
            "Progress file integrity check failed" \
            "task_id=$task_id"
        return 1
    fi

    printf '%s' "$content"
}
```

#### M5.3: Safe File Operations

```bash
# Progress directory MUST be protected
MAINFRAME_PROGRESS_DIR="${MAINFRAME_PROGRESS_DIR:-${TMPDIR:-/tmp}/mainframe_progress}"

_progress_validate_task_id() {
    local task_id="$1"

    # Reject empty
    [[ -z "$task_id" ]] && return 1

    # Reject path traversal
    [[ "$task_id" == *".."* ]] && return 1
    [[ "$task_id" == *"/"* ]] && return 1

    # Allow only safe characters
    [[ "$task_id" =~ ^[a-zA-Z0-9_-]+$ ]] || return 1

    # Length limit
    [[ ${#task_id} -le 64 ]] || return 1

    return 0
}
```

#### M5.5: Quota Enforcement

```bash
# MUST prevent disk exhaustion
readonly PROGRESS_MAX_FILES="${PROGRESS_MAX_FILES:-100}"
readonly PROGRESS_MAX_SIZE_KB="${PROGRESS_MAX_SIZE_KB:-1024}"  # 1MB total
readonly PROGRESS_FILE_MAX_KB="${PROGRESS_FILE_MAX_KB:-10}"    # 10KB per file

_progress_enforce_quota() {
    local dir="$MAINFRAME_PROGRESS_DIR"

    # Count files
    local count
    count=$(find "$dir" -maxdepth 1 -name "*.progress" | wc -l)
    if (( count > PROGRESS_MAX_FILES )); then
        # Remove oldest
        find "$dir" -maxdepth 1 -name "*.progress" -printf '%T@ %p\n' | \
            sort -n | head -n $((count - PROGRESS_MAX_FILES + 10)) | \
            cut -d' ' -f2- | xargs rm -f 2>/dev/null
    fi

    # Check total size
    local size_kb
    size_kb=$(du -sk "$dir" 2>/dev/null | cut -f1)
    if (( size_kb > PROGRESS_MAX_SIZE_KB )); then
        # Remove oldest until under limit
        while (( $(du -sk "$dir" | cut -f1) > PROGRESS_MAX_SIZE_KB / 2 )); do
            find "$dir" -maxdepth 1 -name "*.progress" -printf '%T@ %p\n' | \
                sort -n | head -1 | cut -d' ' -f2- | xargs rm -f
        done
    fi
}
```
